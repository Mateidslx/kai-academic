Persistance des données et boucle canonique de lecture

FILE* f = fopen("donnees.txt", "r");
if (f == NULL) {
    perror("Erreur ouverture fichier");
    return 1;
}

char ligne[256];
while (fgets(ligne, sizeof(ligne), f) != NULL) {
    // Traitement de chaque ligne
    printf("Lue : %s", ligne);
}

fclose(f); // Fermeture obligatoire du descripteur de fichier
