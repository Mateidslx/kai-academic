1. Anatomie d'un Script Bash Robuste 

#!/bin/bash
# Script exemple conforme exigences M. Roose
set -euo pipefail

# Verification du nombre d'arguments
if [ $# -lt 1 ]; then
    echo "Usage: $0 <repertoire_ou_fichier>" >&2
    exit 1
fi

cible="$1"
if [ ! -e "$cible" ]; then
    echo "Erreur : La cible '$cible' n'existe pas." >&2
    exit 2
fi

echo "Cible '$cible' validee avec succes !"
exit 0

2. Les Variables Spéciales Fondamentales

$0
Nom du script exécuté
$1, $2, ...
Arguments positionnels
$#
Nombre d'arguments reçus
$?
