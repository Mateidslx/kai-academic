 3. MODÈLE 1 : L'ÉCHANGE DE DEUX VALEURS
================================================================================

Contexte : Permuter le contenu de deux variables sans perte d'information.
Nécessite impérativement une variable temporaire intermédiaire.

--------------------------------------------------------------------------------
 3.1 Version dans le main :
--------------------------------------------------------------------------------
// val1, val2 >> Échanger deux entiers >> val1, val2
tempVal1 = val1;
val1 = val2;
val2 = tempVal1;

--------------------------------------------------------------------------------
 3.2 Version Sous-Programme officiel (Procédure avec passage par référence) :
--------------------------------------------------------------------------------
void echanger(int& val1, int& val2)
// But : Échange le contenu de val1 et val2
{
    // VARIABLES
    int copieVal1; // Stockage intermédiaire

    // TRAITEMENTS
    copieVal1 = val1;
    val1 = val2;
    val2 = copieVal1;
}


================================================================================
 4. MODÈLE 2 : LA SAISIE-VÉRIFICATION D'UNE VALEUR
================================================================================

Contexte : Récupérer au clavier une valeur valide en obligeant l'utilisateur à
ressaisir tant que la donnée est en dehors de l'intervalle autorisé.
Structure officielle retenue par M. Etcheverry : boucle infinie 'for(;;)' avec
test conditionnel et 'break' interne.

--------------------------------------------------------------------------------
 4.1 Saisie-vérification bornée avec message d'erreur :
--------------------------------------------------------------------------------
#include <iostream>
using namespace std;

int main(void)
{
    // CONSTANTES
    const short int NOTE_MIN = 0;   // Note minimale légale
    const short int NOTE_MAX = 20;  // Note maximale légale

    // VARIABLES
    float note; // Note entrée au clavier

    // TRAITEMENTS
    // (clavier) >> Saisir-vérifier la note >> note
    for ( ; ; )
    {
        cout << "Entrez une note (" << NOTE_MIN << " a " << NOTE_MAX << ") : ";
        cin >> note;

        if ((note >= NOTE_MIN) && (note <= NOTE_MAX))
        {
            break; // Valeur conforme : sortie de boucle
        }

        cout << "Erreur : la note doit etre comprise entre " 
             << NOTE_MIN << " et " << NOTE_MAX << "." << endl;
    }

    cout << "Note validee : " << note << endl;
    return 0;
}

--------------------------------------------------------------------------------
 4.2 Saisie d'une suite avec sentinelle d'arrêt (Exemple TD4 / mainProf) :
--------------------------------------------------------------------------------
// Arrêt lorsque l'utilisateur tape la valeur sentinelle 999
for ( ; ; )
{
    cout << "Entrez une note (999 pour arreter) : ";
    cin >> valeurSaisie;

    if (valeurSaisie == VAL_ARRET)
    {
        break; // L'utilisateur demande la fin de saisie
    }

    if ((valeurSaisie >= NOTE_MIN) && (valeurSaisie <= NOTE_MAX))
    {
        sommeNotes += valeurSaisie;
        nbNotesValides++;
    }
    else
    {
        cout << "Valeur invalide ignoree !" << endl;
    }
}

 5. MODÈLE 3 : PARCOURS COMPLET AVEC TRAITEMENT SYSTÉMATIQUE
================================================================================

Contexte : Examiner CHAQUE case d'un tableau d'effectif connu de manière 
inconditionnelle. Utilise OBLIGATOIREMENT une boucle 'for'.

--------------------------------------------------------------------------------
 5.1 Somme et Moyenne d'un tableau (avec protection division par zéro) :
--------------------------------------------------------------------------------
// tabNotes, EFFECTIF >> Calculer somme et moyenne >> somme, moyenne
somme = 0.0f;
for (i = 0; i < EFFECTIF; i++)
{
    somme += tabNotes[i];
}

if (EFFECTIF > 0)
{
    moyenne = somme / EFFECTIF;
    cout << "Moyenne : " << moyenne << endl;
}
else
{
    cout << "Impossible : effectif nul." << endl;
}

--------------------------------------------------------------------------------
 5.2 Parcours à rebours (De la dernière case à la première) :
--------------------------------------------------------------------------------
// tab, EFFECTIF >> Afficher à rebours >> (écran)
for (i = EFFECTIF; i > 0; i--)
{
    cout << "Case [" << (i - 1) << "] = " << tab[i - 1] << endl;
}

--------------------------------------------------------------------------------
 5.3 Parcours systématique d'une Matrice 2D (Lignes x Colonnes) :
--------------------------------------------------------------------------------
for (ligne = 0; ligne < NB_LIGNES; ligne++)
{
    for (colonne = 0; colonne < NB_COLONNES; colonne++)
    {
        cout << matrice[ligne][colonne] << "\t";
    }
    cout << endl;
}

 6. MODÈLE 4 : PARCOURS COMPLET AVEC TRAITEMENT CONDITIONNÉ
================================================================================

Contexte : Examiner l'intégralité d'un tableau, mais n'appliquer l'action
(comptage, cumul, affichage) QUE sur les éléments qui satisfont un critère.

--------------------------------------------------------------------------------
 6.1 Compter et cumuler les notes supérieures ou égales à 10 :
--------------------------------------------------------------------------------
// tabNotes, EFFECTIF, SEUIL >> Compter et moyenner les admis >> nbAdmis, moyenneAdmis
nbAdmis = 0;
sommeAdmis = 0.0f;

for (i = 0; i < EFFECTIF; i++)
{
    if (tabNotes[i] >= SEUIL_ADMISSION)
    {
        nbAdmis++;
        sommeAdmis += tabNotes[i];
    }
}

if (nbAdmis > 0)
{
    moyenneAdmis = sommeAdmis / nbAdmis;
}

--------------------------------------------------------------------------------
 6.2 Filtrer les voyelles dans une chaîne de caractères :
--------------------------------------------------------------------------------
nbVoyelles = 0;
for (i = 0; i < mot.length(); i++)
{
    c = tolower(mot[i]);
    if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' || c == 'y')
    {
        nbVoyelles++;
    }
}

 7. MODÈLE 5 : RECHERCHE DE PREMIÈRE OCCURRENCE
================================================================================

Contexte : Trouver la PREMIÈRE case vérifiant une condition et s'arrêter
IMMÉDIATEMENT dès qu'elle est trouvée (parcours partiel).

🚨 RÈGLES FORMELLES STRICTES DE M. ETCHEVERRY :
1. La structure DOIT être une boucle 'while'.
2. Utiliser un drapeau booléen explicite 'trouve' initialisé à 'false'.
3. Double condition d'arrêt : 'while (!trouve && (i < EFFECTIF))'.
4. En cas de non-trouvaille sur la case courante, incrémenter l'indice 'i++'.
5. INTERDICTION ABSOLUE de faire un 'break' dans une boucle 'for'.

--------------------------------------------------------------------------------
 7.1 Recherche de valeur dans un tableau d'entiers :
--------------------------------------------------------------------------------
// tab, EFFECTIF, valeurCible >> Trouver la 1re occurrence >> trouve, indiceTrouve
trouve = false;
i = 0;

while (!trouve && (i < EFFECTIF))
{
    if (tab[i] == valeurCible)
    {
        trouve = true; // Arrêt immédiat de la boucle au prochain test
    }
    else
    {
        i++; // On avance à la case suivante
    }
}

if (trouve)
{
    indiceTrouve = i;
    cout << "Valeur trouvee a l'indice : " << indiceTrouve << endl;
}
else
{
    cout << "Valeur non trouvee dans le tableau." << endl;
}

 8. MODÈLE 6 : RECHERCHE D'EXTREMUM (MINIMUM OU MAXIMUM)
================================================================================

Contexte : Trouver la plus grande (ou plus petite) valeur d'un tableau et son indice.

🚨 PIÈGE ÉLIMINATOIRE :
NE JAMAIS initialiser le max à 0 ou le min à 99999 !
Si le tableau contient uniquement des températures négatives (-15, -8, -22), 
un max initialisé à 0 produira un résultat totalement faux !

--------------------------------------------------------------------------------
 8.1 Recherche du Maximum et de sa position :
--------------------------------------------------------------------------------
// tab, EFFECTIF >> Déterminer le max et sa position >> valeurMax, indiceMax
valeurMax = tab[0]; // Règle d'or : initialiser avec le PREMIER élément
indiceMax = 0;

// La boucle commence strictement à l'indice 1
for (i = 1; i < EFFECTIF; i++)
{
    if (tab[i] > valeurMax)
    {
        valeurMax = tab[i];
        indiceMax = i;
    }
}


================================================================================
 9. MODÈLE 7 : RECHERCHE DE DERNIÈRE OCCURRENCE
================================================================================

Contexte : Trouver la DERNIÈRE fois qu'une valeur apparaît.
Stratégie optimale : Parcourir à rebours depuis 'EFFECTIF - 1' jusqu'à '0'.
Dès que la valeur est trouvée, on s'arrête (première trouvaille en partant de la fin).

--------------------------------------------------------------------------------
 9.1 Implémentation canonique :
--------------------------------------------------------------------------------
// tab, EFFECTIF, cible >> Trouver la dernière occurrence >> trouve, indiceDernier
trouve = false;
if (EFFECTIF > 0)
{
    i = EFFECTIF - 1;
    for ( ; ; )
    {
        if (tab[i] == cible)
        {
            trouve = true;
            indiceDernier = i;
            break;
        }

        if (i == 0)
        {
            break; // Tout le tableau a été inspecté
        }
        i--;
    }
}


================================================================================
 10. MODÈLE 8 : CONTRÔLE DE PROPRIÉTÉ GLOBALE (DÉTECTION D'ANOMALIE)
================================================================================

Contexte : Vérifier si TOUS les éléments respectent une règle (ex: tableau trié).
Principe : On suppose que la règle est vraie ('valide = true'). On cherche 
la PREMIÈRE anomalie qui contredit la règle. Dès qu'une anomalie est trouvée, 
on s'arrête ('valide = false').

--------------------------------------------------------------------------------
 10.1 Vérifier si un tableau est strictement trié croissant :
--------------------------------------------------------------------------------
// tab, EFFECTIF >> Vérifier si le tableau est trié >> estTrie
estTrie = true;
i = 0;

// Borne d'arrêt à EFFECTIF - 1 car on compare tab[i] et tab[i+1]
while (estTrie && (i < EFFECTIF - 1))
{
    if (tab[i] > tab[i + 1])
    {
        estTrie = false; // Anomalie détectée : rupture d'ordre croissant
    }
    else
    {
        i++;
    }
}

--------------------------------------------------------------------------------
 10.2 Palindrome à deux curseurs convergents (Début et Fin) :
--------------------------------------------------------------------------------
// mot >> Vérifier si le mot est un palindrome >> estPalindrome
estPalindrome = true;
if (mot.length() > 0)
{
    gauche = 0;
    droite = mot.length() - 1;

    while (estPalindrome && (gauche < droite))
    {
        if (mot[gauche] != mot[droite])
        {
            estPalindrome = false; // Différence trouvée : pas un palindrome
        }
        else
        {
            gauche++;
            droite--;
        }
    }
}

 11. MODÈLE 9 : ALGORITHMES AVANCÉS (CEINTURE NOIRE 20/20)
================================================================================

--------------------------------------------------------------------------------
 11.1 Tri Pair-Impair par Transposition (Odd-Even Transposition Sort) :
--------------------------------------------------------------------------------
Contexte officiel : Épreuve du Défi Individuel S1.01 de Patrick Etcheverry.
Algorithme de tri par comparaison en deux phases alternées (Phase Impaire / Phase Paire).
Le tri continue tant qu'au moins une permutation a eu lieu durant la passe.

void trierParTpDecroissant(Etudiant tab[], unsigned int nbCases)
// But : Trier un tableau d'étudiants par numéro de TP décroissant
{
    // VARIABLES
    bool triTermine;     // Vrai si aucune permutation n'a eu lieu
    unsigned int i;      // Indice de parcours
    Etudiant temp;       // Variable d'échange

    // TRAITEMENTS
    triTermine = false;

    while (!triTermine)
    {
        triTermine = true;

        // 1. Passe impaire : comparer cases (1,2), (3,4), (5,6)...
        for (i = 1; i + 1 < nbCases; i += 2)
        {
            if (tab[i].tp < tab[i + 1].tp) // Ordre décroissant voulu
            {
                temp = tab[i];
                tab[i] = tab[i + 1];
                tab[i + 1] = temp;
                triTermine = false;
            }
        }

        // 2. Passe paire : comparer cases (0,1), (2,3), (4,5)...
        for (i = 0; i + 1 < nbCases; i += 2)
        {
            if (tab[i].tp < tab[i + 1].tp) // Ordre décroissant voulu
            {
                temp = tab[i];
                tab[i] = tab[i + 1];
                tab[i + 1] = temp;
                triTermine = false;
            }
        }
    }
}

 12. GUIDE OFFICIEL DES SOUS-PROGRAMMES (PROLOGUE R1.01)
================================================================================

--------------------------------------------------------------------------------
 12.1 Fonction vs Procédure : la règle absolue
--------------------------------------------------------------------------------
- FONCTION :
  * Renvoie EXACTEMENT UN SEUL résultat typé par 'return'.
  * Ne modifie aucun de ses paramètres.
  * Zéro affichage, zéro saisie (aucun effet de bord).
  * S'utilise directement dans une expression : 'if (estMajeur(age)) ...'
- PROCÉDURE ('void') :
  * 0 résultat, ou PLUSIEURS résultats transmis par référence '&'.
  * Ou action modifiant l'état (affichage, saisie, modification de tableau).

--------------------------------------------------------------------------------
 12.2 Ordre officiel des paramètres (Norme Etcheverry)
--------------------------------------------------------------------------------
Toujours ordonner les paramètres formels selon :
1. Données (in)
2. Données / Résultats (inout)
3. Résultats (out)

Pour un tableau : TOUJOURS passer le tableau puis sa taille :
void afficherTableau(const int tab[], unsigned short int nbCases);

--------------------------------------------------------------------------------
 12.3 Modes de transmission des paramètres
--------------------------------------------------------------------------------
1. PAR VALEUR (ex: 'int age', 'float note') :
   Pour toute donnée primitive en lecture seule dont la taille est <= 8 octets.
2. PAR RÉFÉRENCE CONSTANTE (ex: 'const string& nom', 'const Etudiant& etu') :
   Pour toute donnée complexe en lecture seule (taille >= 16 octets) afin
   d'éviter une copie mémoire inutile.
3. PAR RÉFÉRENCE (ex: 'int& resultat', 'bool& trouve') :
   Pour tout paramètre qui doit être modifié ou qui constitue un résultat retourné.
4. TABLEAU EN PARAMÈTRE :
   - Tableau en lecture seule : 'const int tab[], unsigned short int nbCases'
   - Tableau modifié sur place : 'int tab[], unsigned short int nbCases'

 13. STRUCTURE COMPOSITE CANONIQUE (struct Etudiant)
================================================================================

struct Etudiant
{
    string nom;                    // Nom de famille de l'étudiant
    string prenom;                 // Prénom de l'étudiant
    unsigned short int td;         // Numéro de groupe de TD (1, 2, 3...)
    unsigned short int tp;         // Numéro de groupe de TP (1 à 6)
    float moyenne;                 // Moyenne générale semestrielle
};

// Recherche d'un étudiant par son nom (Modèle 5 sur struct) :
bool rechercherEtudiantParNom(const Etudiant promo[], 
                             unsigned short int effectif, 
                             const string& nomRecherche, 
                             unsigned short int& indiceTrouve)
{
    // VARIABLES
    bool trouve;
    unsigned short int i;

    // TRAITEMENTS
    trouve = false;
    i = 0;

    while (!trouve && (i < effectif))
    {
        if (promo[i].nom == nomRecherche)
        {
            trouve = true;
            indiceTrouve = i;
        }
        else
        {
            i++;
        }
    }

    return trouve;
}
